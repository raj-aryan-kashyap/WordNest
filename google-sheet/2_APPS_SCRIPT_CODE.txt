/**
 * ============================================================
 *  WordNest sync server
 *  Google Apps Script + Google Sheet. Free, no other services.
 * ============================================================
 *
 *  SETUP (about 5 minutes)
 *  1. Open the WordNest Sync Google Sheet.
 *  2. Extensions > Apps Script. Delete the sample code, paste this file, Save.
 *  3. Deploy > New deployment > Select type: Web app.
 *       Execute as:      Me
 *       Who has access:  Anyone
 *  4. Approve the permissions, then copy the Web app URL (it ends in /exec).
 *  5. Paste the URL into js/config.js as SYNC_URL in the app, and push to GitHub.
 *
 *  UPDATING THIS SCRIPT LATER
 *  Deploy > Manage deployments > pencil icon > Version: New version > Deploy.
 *  This keeps the same URL, so the app keeps working.
 *
 *  TABS IT USES
 *  Users  One row per person. Readable totals, then the app's raw data.
 *  Words    One row per word per person, rewritten on every sync. Read only.
 *  Library  Master list of every word anyone has seen, with its full card,
 *           sorted by context then newest. New words are added on each sync,
 *           and words a person hasn't seen yet are sent back to their app.
 *  Missing tabs are created and formatted automatically. A new Library tab
 *  starts filled with the app's built-in starter words.
 *
 *  SMART WORD PICKS (optional, free)
 *  Get a free key at https://aistudio.google.com/apikey, then in the sheet
 *  click WordNest > Set Gemini key. The key is stored in this script's
 *  properties, never on phones. Calls are capped (see AI settings below).
 *
 *  SECURITY
 *  The app sends a hash of name + PIN, never the PIN itself. Good enough for
 *  a personal project. Don't reuse an important PIN.
 */

/** Visiting the URL in a browser shows this. Handy to check the deployment works. */
function doGet() {
  return out({ ok: true, app: 'WordNest', message: 'Sync is running.' });
}

/** All app requests arrive here as POST with a JSON body. */
function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    var req = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    var user = String(req.user || '').trim().toLowerCase();
    var pin = String(req.pin || '');

    if (!/^[a-z0-9._-]{2,40}$/.test(user)) return out({ ok: false, code: 'bad_user', error: 'Use 2 to 40 letters or numbers for your name.' });
    if (!/^[a-f0-9]{64}$/.test(pin)) return out({ ok: false, code: 'bad_pin', error: 'That PIN was not sent correctly.' });

    var sheet = getUsersSheet();
    var row = findRow(sheet, user);

    // Sign in, or create the account if the name is new.
    if (req.action === 'login') {
      if (row === -1) {
        sheet.appendRow([user, pin, new Date(), 0, 0, 0, 0, 0, '', 0]);
        return out({ ok: true, created: true });
      }
      if (sheet.getRange(row, COL.PIN).getValue() !== pin) return out({ ok: false, code: 'wrong_pin', error: 'That PIN is wrong for this name.' });
      return out({ ok: true, created: false });
    }

    if (row === -1) return out({ ok: false, code: 'no_user', error: 'Account not found. Sign in again.' });
    if (sheet.getRange(row, COL.PIN).getValue() !== pin) return out({ ok: false, code: 'wrong_pin', error: 'That PIN is wrong for this name.' });

    // Merge the phone's data with the saved copy, save, and send the result back.
    if (req.action === 'suggest') {
      lock.releaseLock(); // Gemini can take a few seconds; don't block other syncs
      return out(suggestWords(user, req));
    }

    if (req.action === 'ai_status') return out({ ok: true, ai: aiUsage() });

    if (req.action === 'sync') {
      var merged = mergeSnap(req.data || null, readData(sheet, row));
      writeData(sheet, row, merged);
      writeSummary(sheet, row, merged);
      if (WRITE_WORDS_TAB) writeWords(user, merged);
      var offer = [];
      try {
        writeLibrary(user, req.library || []);
        offer = libraryOffer(merged, req.contexts || []);
      } catch (libErr) {
        console.warn('Library skipped: ' + libErr); // sync itself still succeeds
      }
      var ai = null;
      try { ai = aiUsage(); } catch (aiErr) { /* never let usage info break a sync */ }
      return out({ ok: true, data: merged, offer: offer, ai: ai });
    }

    return out({ ok: false, code: 'bad_action', error: 'Unknown action.' });
  } catch (err) {
    return out({ ok: false, code: 'server', error: 'Sync server error: ' + String((err && err.message) || err) });
  } finally {
    try { lock.releaseLock(); } catch (ignore) {}
  }
}

    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=AND($A2<>"",MOD(ROW(),2)=0)')
      .setBackground(C.band).setRanges([all]).build()
  ]);
  sheet.setTabColor(C.ink);
}

function findRow(sheet, user) {
  var last = sheet.getLastRow();
  if (last < 2) return -1;
  var values = sheet.getRange(2, COL.USER, last - 1, 1).getValues();
  for (var i = 0; i < values.length; i++) {
    if (String(values[i][0]).toLowerCase() === user) return i + 2;
  }
  return -1;
}

/** The app's data is JSON split across cells from column K. Each part starts with "~". */
function readData(sheet, row) {
  var n = Number(sheet.getRange(row, COL.PARTS).getValue()) || 0;
  if (!n) return null;
  var parts = sheet.getRange(row, COL.DATA, 1, n).getValues()[0];
  var json = parts.map(function (p) { return String(p).replace(/^~/, ''); }).join('');
  try { return JSON.parse(json); } catch (e) { return null; }
}

function writeData(sheet, row, data) {
  var json = JSON.stringify(data || {});
  var parts = [];
  for (var i = 0; i < json.length; i += CHUNK) parts.push('~' + json.slice(i, i + CHUNK));
  var lastCol = sheet.getLastColumn();
  if (lastCol >= COL.DATA) sheet.getRange(row, COL.DATA, 1, lastCol - COL.DATA + 1).clearContent();
  sheet.getRange(row, COL.LAST_SYNC).setValue(new Date());
  sheet.getRange(row, COL.PARTS).setValue(parts.length);
  if (parts.length) {
    ensureSize(sheet, row, COL.DATA + parts.length - 1);
    var range = sheet.getRange(row, COL.DATA, 1, parts.length);
    range.setNumberFormat('@');
    range.setValues([parts]);
    // Label any new data columns so the header row stays tidy.
    for (var c = 0; c < parts.length; c++) {
      var h = sheet.getRange(1, COL.DATA + c);
      if (!h.getValue()) h.setValue('App data ' + (c + 1)).setBackground(C.raw).setFontColor(C.inkText).setFontWeight('bold');
    }
  }
}

/** Readable totals for the person's row. */
function writeSummary(sheet, row, snap) {
  var s = (snap && snap.state) || {};
  var counts = { learning: 0, mastered: 0, known: 0 };
  Object.keys(s.words || {}).forEach(function (id) {
    var st = s.words[id].s;
    if (counts[st] !== undefined) counts[st]++;
  });
  var days = (s.days || []).slice().sort();
  var last = days.length ? toDate(days[days.length - 1]) : '';
  sheet.getRange(row, COL.USER).setFontWeight('bold');
  sheet.getRange(row, COL.PIN).setFontColor(C.raw);
  sheet.getRange(row, COL.PARTS, 1, 4).setFontColor(C.raw).setWrap(false);
  sheet.getRange(row, COL.LAST_SYNC).setNumberFormat('d mmm yyyy, hh:mm');
  sheet.getRange(row, COL.LAST_ACTIVE).setNumberFormat('d mmm yyyy');
  sheet.getRange(row, COL.LEARNING, 1, 6).setHorizontalAlignment('center');
  sheet.getRange(row, COL.LEARNING, 1, 6).setValues([[
    counts.learning, counts.mastered, counts.known, streak(days), days.length, last
  ]]);
}

/** Days in a row up to today (or yesterday, if today has no practice yet). */
function streak(days) {
  var set = {};
  days.forEach(function (d) { set[d] = 1; });
  var d = new Date();
  if (!set[key(d)]) d.setDate(d.getDate() - 1);
  var n = 0;
  while (set[key(d)]) { n++; d.setDate(d.getDate() - 1); }
  return n;
}

    if (bold) b.setBold(true);
    return b.build();
  };
  sheet.setConditionalFormatRules([
    rule('=$C2="Learning"', 'C2:C20000', C.learning, C.learningText, true),
    rule('=$C2="Mastered"', 'C2:C20000', C.mastered, C.masteredText, true),
    rule('=$C2="Known"', 'C2:C20000', C.known, C.knownText, false),
    rule('=AND($C2<>"Known",$E2<>"",$E2<=TODAY())', 'E2:E20000', '#FCEFD9', '#8A5A12', true),
    rule('=AND($H2<>"",$H2<0.5)', 'H2:H20000', null, '#B5463B', true),
    rule('=AND($A2<>"",MOD(ROW(),2)=0)', 'A2:J20000', C.band, null, false)
  ]);
  if (!sheet.getFilter()) sheet.getRange(1, 1, sheet.getMaxRows(), WORDS_HEADERS.length).createFilter();
  sheet.setTabColor('#2E8C7E');
}

var STATUS_LABEL = { learning: 'Learning', mastered: 'Mastered', known: 'Known' };
var STATUS_ORDER = { learning: 0, mastered: 1, known: 2 };

function writeWords(user, snap) {
  var sheet = getWordsSheet();
  var words = (snap && snap.state && snap.state.words) || {};
  var last = sheet.getLastRow();
  var others = last >= 2
    ? sheet.getRange(2, 1, last - 1, WORDS_HEADERS.length).getValues().filter(function (r) { return String(r[0]).toLowerCase() !== user; })
    : [];

  var mine = Object.keys(words).map(function (id) { return words[id]; }).sort(function (a, b) {
    return (STATUS_ORDER[a.s] - STATUS_ORDER[b.s]) || String(a.due || '9').localeCompare(String(b.due || '9')) || String(a.w).localeCompare(String(b.w));
  }).map(function (r) {
    return [user, r.w, STATUS_LABEL[r.s] || r.s, r.s === 'known' ? '' : r.box,
      r.s === 'known' ? '' : toDate(r.due), r.ok || 0, r.bad || 0, '', toDate(r.add), toDate(r.last)];
  });

  var rows = others.concat(mine);
  if (last >= 2) sheet.getRange(2, 1, last - 1, WORDS_HEADERS.length).clearContent();
  if (!rows.length) return;
  ensureSize(sheet, rows.length + 1, WORDS_HEADERS.length);
  // Accuracy is a formula so it stays correct if anyone changes Right or Wrong.
  for (var i = 0; i < rows.length; i++) {
    var r = i + 2;
    rows[i][7] = '=IF(F' + r + '+G' + r + '=0,"",F' + r + '/(F' + r + '+G' + r + '))';
  }
  sheet.getRange(2, 1, rows.length, WORDS_HEADERS.length).setValues(rows);
  // Keep formats tidy on rows the sheet has not formatted yet.
  sheet.getRange(2, 5, rows.length, 1).setNumberFormat('d mmm yyyy');
  sheet.getRange(2, 8, rows.length, 1).setNumberFormat('0%');
  sheet.getRange(2, 9, rows.length, 2).setNumberFormat('d mmm yyyy');
  sheet.getRange(2, 3, rows.length, 8).setHorizontalAlignment('center');
  sheet.getRange(2, 2, rows.length, 1).setFontWeight('bold');
}

/** Menu item: add any starter words missing from the Library. Safe to run again. */
function fillStarterLibrary() {
  var sheet = getLibrarySheet();
  var added;
  try {
    added = addStarterRows(sheet);
  } catch (e) {
    var err = 'Could not add starter words: ' + e.message;
    console.error(err);
    SpreadsheetApp.getActive().toast(err, 'WordNest', 15);
    throw e;
  }
  var total = Math.max(0, sheet.getLastRow() - 1);
  var msg = added ? added + ' starter words added. The Library now has ' + total + ' words.'
                  : 'All ' + STARTER.length + ' starter words are already in the Library (' + total + ' words in total).';
  SpreadsheetApp.getActive().toast(msg, 'WordNest', 8);
  console.log(msg);
}

/** The app's built-in words (STARTER, at the end of this file) as Library rows. Returns how many were added. */
function addStarterRows(sheet) {
  var entries = STARTER.map(function (s) {
    var tags = String(s[12]).split(',');
    return { id: s[0], w: s[1], p: s[2], d: s[3], m: s[4], say: s[5], ipa: s[6], past: s[7], present: s[8], future: s[9],
      other: '', hi: s[10], him: s[11], src: 'Starter pack', ctx: TOPIC_NAME[tags[0]] || 'Everyday', t: s[12], added: '' };
  });
  var before = Math.max(0, sheet.getLastRow() - 1);
  writeLibrary('wordnest', entries, true);
  return Math.max(0, sheet.getLastRow() - 1) - before;
}

/**
 * Add new words to the Library. One row per word (matched by text, any case).
 * A word someone else already added gets this person in "Also used by",
 * and empty cells (for example a missing Hindi meaning) are filled in.
 * Then the tab is sorted by Context, newest first, then Word.
 */
function writeLibrary(user, entries, isStarter) {
  if (!entries.length) return;
  var sheet = getLibrarySheet();
  var n = LIB_HEADERS.length;
  var last = sheet.getLastRow();
  var rows = last >= 2 ? sheet.getRange(2, 1, last - 1, n).getValues() : [];
  var index = {};
  rows.forEach(function (r, i) { if (r[L.WORD]) index[String(r[L.WORD]).toLowerCase()] = i; });
  var tz = Session.getScriptTimeZone();
  var changed = false;

  entries.forEach(function (e) {
    var word = String(e.w || '').trim();
    if (!word) return;
    var k = word.toLowerCase();
    var fields = [[L.MEANING, e.m], [L.SAY, e.say], [L.IPA, e.ipa], [L.POS, e.p], [L.PAST, e.past],
      [L.PRESENT, e.present], [L.FUTURE, e.future], [L.OTHER, e.other], [L.HI, e.hi], [L.HIM, e.him]];
    if (k in index) {
      var row = rows[index[k]];
      if (!isStarter && String(row[L.BY]) !== user) {
        var also = String(row[L.ALSO] || '').split(', ').filter(String);
        if (also.indexOf(user) === -1) { also.push(user); row[L.ALSO] = also.join(', '); changed = true; }
      }
      fields.forEach(function (f) { if (!row[f[0]] && f[1]) { row[f[0]] = f[1]; changed = true; } });
      return;
    }
    var added = toDate(e.added) || new Date();
    var r = new Array(n).fill('');
    r[L.CTX] = e.ctx || 'Everyday';
    r[L.WORD] = word;
    r[L.LEVEL] = Math.round(Number(e.d) || 3);
    fields.forEach(function (f) { r[f[0]] = f[1] || ''; });
    r[L.SOURCE] = e.src || '';
    r[L.ADDED] = added;
    r[L.DAY] = Utilities.formatDate(added, tz, 'EEEE');
    r[L.BY] = isStarter ? 'WordNest' : user;
    r[L.TOPICS] = e.t || '';
    r[L.KEY] = e.id || '';
    index[k] = rows.length;
    rows.push(r);
    changed = true;
  });

  if (!changed || !rows.length) return;
  ensureSize(sheet, rows.length + 1, n);
  var range = sheet.getRange(2, 1, rows.length, n);
  range.setValues(rows);
  sheet.getRange(2, L.ADDED + 1, rows.length, 1).setNumberFormat('d mmm yyyy');
  range.sort([{ column: L.CTX + 1, ascending: true }, { column: L.ADDED + 1, ascending: false }, { column: L.WORD + 1, ascending: true }]);
}

/** Library words this person doesn't have yet. Their goals' contexts first, then newest. */
function libraryOffer(snap, contexts) {
  var sheet = getLibrarySheet();
  var last = sheet.getLastRow();
  if (last < 2) return [];
  var have = {};
  var words = (snap && snap.state && snap.state.words) || {};
  Object.keys(words).forEach(function (id) { have[String(words[id].w).toLowerCase()] = 1; });
  var want = {};
  contexts.forEach(function (c) { want[String(c).toLowerCase()] = 1; });

  var rows = sheet.getRange(2, 1, last - 1, LIB_HEADERS.length).getValues().filter(function (r) {
    var key = String(r[L.KEY]);
    return r[L.WORD] && r[L.MEANING] && !have[String(r[L.WORD]).toLowerCase()] && key.indexOf('s:') !== 0;
  });
  rows.sort(function (a, b) {
    var wa = want[String(a[L.CTX]).toLowerCase()] ? 0 : 1;
    var wb = want[String(b[L.CTX]).toLowerCase()] ? 0 : 1;
    return (wa - wb) || (new Date(b[L.ADDED]) - new Date(a[L.ADDED]));
  });
  return rows.slice(0, LIB_OFFER).map(function (r) {
    return { id: r[L.KEY], w: r[L.WORD], d: r[L.LEVEL], m: r[L.MEANING], say: r[L.SAY], ipa: r[L.IPA], p: r[L.POS],
      past: r[L.PAST], present: r[L.PRESENT], future: r[L.FUTURE], other: r[L.OTHER], hi: r[L.HI], him: r[L.HIM],
      t: r[L.TOPICS], ctx: r[L.CTX] };
  });
}

/** Today's usage. The count resets when the day changes in the script's time zone. */
function aiUsage() {

var WORD_SCHEMA = {
  type: 'OBJECT',
  properties: { words: { type: 'ARRAY', items: { type: 'OBJECT', properties: {
    word: { type: 'STRING' }, pos: { type: 'STRING' }, ipa: { type: 'STRING' }, say: { type: 'STRING' },
    meaning: { type: 'STRING' }, note: { type: 'STRING' },
    present: { type: 'STRING' }, past: { type: 'STRING' }, future: { type: 'STRING' },
    hindi: { type: 'STRING' }, hindiMeaning: { type: 'STRING' },
    level: { type: 'NUMBER' }, goals: { type: 'ARRAY', items: { type: 'STRING' } }
  }, required: ['word', 'meaning', 'present', 'past', 'future', 'hindi', 'level', 'goals'] } } },
  required: ['words']
};

/** One Gemini call. Falls back to the lighter model if the main one is missing. Throws Error with .code. */
function callGemini(apiKey, prompt, schema) {
  var models = [props().getProperty('GEMINI_MODEL') || AI.MODEL, AI.FALLBACK_MODEL];
  var lastErr;
  for (var i = 0; i < models.length; i++) {
    var cfg = { responseMimeType: 'application/json', temperature: 0.8 };
    if (schema) cfg.responseSchema = schema;
    var res = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + models[i] + ':generateContent', {
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { 'x-goog-api-key': apiKey },
      payload: JSON.stringify({ contents: [{ role: 'user', parts: [{ text: prompt }] }], generationConfig: cfg })
    });
    var code = res.getResponseCode();
    if (code === 200) {
      var data = JSON.parse(res.getContentText());
      var parts = (((data.candidates || [])[0] || {}).content || {}).parts || [];
      var text = parts.map(function (p) { return p.text || ''; }).join('').replace(/```json|```/g, '').trim();
      return { json: JSON.parse(text), model: models[i] };
    }
    lastErr = new Error(code === 429 ? 'Too many requests right now.' : code === 400 || code === 403 ? 'The key was refused. Check it in AI Studio.' : 'Gemini error ' + code + '.');
    lastErr.code = code === 429 ? 'ai_busy' : code === 404 ? 'ai_model' : 'ai_error';
    if (code !== 404) break; // only a missing model is worth retrying with the other one
  }
  throw lastErr;
}

/**
 * Action "suggest": the app sends a prompt built from the learner's behaviour
 * and the words it has seen. We add the Library's words to the avoid list,
 * call Gemini, keep only good words, add them to the Library, and return them.
 */
function suggestWords(user, req) {
  var p = props();
  var apiKey = p.getProperty('GEMINI_API_KEY');
  if (!apiKey) return { ok: false, code: 'no_key', error: 'Smart picks are off. Add a Gemini key in the sheet.', ai: aiUsage() };

  var usage = aiUsage();
  if (usage.used >= usage.limit) return { ok: false, code: 'ai_limit', error: "Today's smart picks are used up. More tomorrow.", ai: usage };
  var now = Date.now();
  var cool = Number(p.getProperty('ai_cooldown') || 0);
  if (now < cool) return { ok: false, code: 'ai_busy', retryIn: Math.ceil((cool - now) / 1000), error: 'Gemini is busy. Trying again soon.', ai: usage };
  var last = Number(p.getProperty('ai_last') || 0);
  if (now - last < AI.MIN_GAP_SECONDS * 1000) return { ok: false, code: 'ai_wait', retryIn: Math.ceil((AI.MIN_GAP_SECONDS * 1000 - (now - last)) / 1000), ai: usage };
  p.setProperty('ai_last', String(now));

  // Words to avoid: what the app sent, plus everything already in the Library.
  var avoid = {};
  (req.avoid || []).forEach(function (w) { avoid[String(w).toLowerCase()] = 1; });
  var lib = getLibrarySheet();
  var lastRow = lib.getLastRow();
  var libWords = lastRow >= 2 ? lib.getRange(2, L.WORD + 1, lastRow - 1, 1).getValues().map(function (r) { return String(r[0]); }) : [];
  libWords.forEach(function (w) { avoid[w.toLowerCase()] = 1; });
  var prompt = String(req.prompt || '').slice(0, 24000) +
    (libWords.length ? '\nAlso never suggest these words, they are already in the shared library: ' + libWords.slice(-600).join(', ') : '');

  var result;
  try {
    result = callGemini(apiKey, prompt, WORD_SCHEMA);
  } catch (e) {
    if (e.code === 'ai_busy') p.setProperty('ai_cooldown', String(now + AI.COOLDOWN_MINUTES * 60000));
    return { ok: false, code: e.code || 'ai_error', error: e.message, ai: aiUsage() };
  }
  var today = key(new Date());
  if (p.getProperty('ai_day') !== today) { p.setProperty('ai_day', today); p.setProperty('ai_used', '0'); }
  p.setProperty('ai_used', String(Number(p.getProperty('ai_used') || 0) + 1));

  var goalTags = req.goalTags || {};
  var cards = cleanAiWords((result.json && result.json.words) || [], avoid, goalTags, req.contextNames || {});

  // Add them to the Library straight away (one lock, like a sync).
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    writeLibrary(user, cards.map(function (c) {
      return { id: c.id, w: c.w, p: c.p, d: c.d, m: c.m, say: c.say, ipa: c.ipa, past: c.past, present: c.present, future: c.future,
        other: '', hi: c.hi, him: c.him, src: 'AI (Gemini)', ctx: c.ctx, t: c.t.join(','), added: today };
    }));
  } catch (e) {
    console.warn('Library write skipped: ' + e);
  } finally {
    try { lock.releaseLock(); } catch (ignore) {}
  }
  return { ok: true, cards: cards, model: result.model, ai: aiUsage() };
}

/** Keep only words that are new, complete and actually used in their examples. */
function cleanAiWords(list, avoid, goalTags, contextNames) {
  var out = [];
  list.forEach(function (x) {
    var w = String(x.word || '').trim();
    var k = w.toLowerCase();
    if (!w || w.length > 40 || /\d/.test(w) || avoid[k]) return;
    var m = String(x.meaning || '').trim();
    if (!m || m.split(/\s+/).length > 25) return;
    var stem = k.split(/[\s-]+/)[0].replace(/(e|y)$/, '').slice(0, 6);
    var ex = [x.past, x.present, x.future].map(function (s) { return String(s || '').trim(); });
    var uses = ex.filter(function (s) { return s && s.toLowerCase().indexOf(stem) !== -1; }).length;
    if (uses < 2) return; // examples must really use the word
    var hi = String(x.hindi || '').trim();
    if (!/[\u0900-\u097F]/.test(hi)) hi = '';
    var ids = (x.goals || []).map(String);
    var tags = [];
    ids.forEach(function (id) { (goalTags[id] || [id]).forEach(function (t) { if (tags.indexOf(t) === -1) tags.push(t); }); });
    if (!tags.length) tags = ['general'];
    avoid[k] = 1;
    out.push({
      id: 'a:' + k.replace(/\s+/g, '-').replace(/[^a-z0-9-]/g, ''), w: w, p: String(x.pos || ''), ipa: String(x.ipa || ''),
      say: String(x.say || ''), m: m, note: String(x.note || '').trim(), past: ex[0], present: ex[1], future: ex[2],
      hi: hi, him: hi ? String(x.hindiMeaning || '').trim() : '', d: Math.max(1, Math.min(5, Number(x.level) || 3)),
      t: tags, ctx: contextNames[ids[0]] || TOPIC_NAME[tags[0]] || 'Everyday'
    });
  });
  return out;
}

/**
 * Make sure the tab has room for `rows` x `cols` cells. Sheets imported from
 * Excel often have only as many rows as their data, and writing past the edge
 * fails with "coordinates are outside the dimensions of the sheet".
 */
function ensureSize(sheet, rows, cols) {
  var maxR = sheet.getMaxRows();
  if (maxR < rows) sheet.insertRowsAfter(maxR, rows - maxR + 200);
  var maxC = sheet.getMaxColumns();
  if (cols && maxC < cols) sheet.insertColumnsAfter(maxC, cols - maxC);
}

/** Run formatting without letting a styling problem stop the data from being written. */
function safely(fn) {
  try { fn(); } catch (e) { console.warn('Formatting skipped: ' + e); }
}

function key(d) {
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');

/** "2026-10-05" to a real date, so the sheet can sort and format it. */
function toDate(dayKey) {
  if (!dayKey || !/^\d{4}-\d{2}-\d{2}$/.test(dayKey)) return '';
  var p = dayKey.split('-');
  return new Date(Number(p[0]), Number(p[1]) - 1, Number(p[2]));
}

/* ---------- Merge (keep in step with mergeSnap() in js/sync.js) ---------- */

/**
 * Combines two copies of a person's data:
 *  words    per word, the newest change wins
 *  history  joined, duplicates removed, last 1,500 kept
 *  days     joined (for the streak)
 *  reviews  a finished daily revision is never lost
 *  profile and learner model: the newest one wins
 */
function mergeSnap(a, b) {
  if (!a || !a.state) return b;
  if (!b || !b.state) return a;
  var A = a.state, B = b.state;
  var newer = function (x, y) { return ((y && y.u) || 0) > ((x && x.u) || 0) ? y : x; };

  var state = Object.assign({}, A);
  state.profile = Object.assign({}, newer(A.profile, B.profile));
  state.profile.onboarded = !!((A.profile && A.profile.onboarded) || (B.profile && B.profile.onboarded));
  state.model = newer(A.model, B.model);

  var words = Object.assign({}, A.words || {});
  Object.keys(B.words || {}).forEach(function (id) {
    var bw = B.words[id], aw = words[id];
    if (!aw || (bw.u || 0) > (aw.u || 0)) words[id] = bw;
  });
  state.words = words;

  var seen = {}, history = [];
  (A.history || []).concat(B.history || []).forEach(function (h) {
    var k = h.t + '|' + h.id + '|' + h.k;
    if (!seen[k]) { seen[k] = 1; history.push(h); }
  });
  history.sort(function (x, y) { return x.t - y.t; });
  state.history = history.slice(-1500);

  var days = {};
  (A.days || []).concat(B.days || []).forEach(function (d) { days[d] = 1; });
  state.days = Object.keys(days).sort();

  var reviews = Object.assign({}, B.reviews || {}, A.reviews || {});
  Object.keys(B.reviews || {}).forEach(function (d) {
    if (B.reviews[d].done && !(A.reviews && A.reviews[d] && A.reviews[d].done)) reviews[d] = B.reviews[d];
  });
  state.reviews = reviews;

  delete state.sync;
  return { state: state, cards: Object.assign({}, b.cards || {}, a.cards || {}) };
}
